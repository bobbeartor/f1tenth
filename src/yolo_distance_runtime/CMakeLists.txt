cmake_minimum_required(VERSION 3.18)
project(yolo_distance_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MULTICLASS_YOLO_TENSORRT "Use existing TensorRT/CUDA YOLO backend" ON)
set(YOLO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../multiclass_yolo")
find_package(ament_cmake REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/DepthaiWorkspace.cmake")
find_package(ament_index_cpp REQUIRED)
find_package(point_cloud REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core dnn highgui imgproc imgcodecs)
# Initialize CUDA before creating its target: CUDA_ARCHITECTURES is copied
# from CMAKE_CUDA_ARCHITECTURES at target creation, not at target_sources().
if(MULTICLASS_YOLO_TENSORRT)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()
add_library(multiclass_detector STATIC ${YOLO_SOURCE_DIR}/src/yolox_detector.cpp)
target_include_directories(multiclass_detector PUBLIC ${YOLO_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(multiclass_detector PUBLIC ${OpenCV_LIBS})
if(MULTICLASS_YOLO_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu /usr/local/TensorRT/include)
  find_library(TENSORRT_LIBRARY nvinfer HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  find_library(TENSORRT_PARSER nvonnxparser HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  find_library(TENSORRT_PLUGIN nvinfer_plugin HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT TENSORRT_PARSER OR NOT TENSORRT_PLUGIN)
    message(FATAL_ERROR "TensorRT development headers/libraries required; see README.md")
  endif()
  target_sources(multiclass_detector PRIVATE ${YOLO_SOURCE_DIR}/src/tensorrt_yolox_backend.cpp ${YOLO_SOURCE_DIR}/src/cuda_yolox_preprocessor.cu)
  target_include_directories(multiclass_detector PRIVATE ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(multiclass_detector PRIVATE CUDA::cudart ${TENSORRT_LIBRARY} ${TENSORRT_PARSER} ${TENSORRT_PLUGIN})
else()
  target_sources(multiclass_detector PRIVATE ${YOLO_SOURCE_DIR}/src/tensorrt_disabled.cpp)
endif()
# Reuse algorithms and camera integration; no roi_lidar/BEV/manual package dependency.
add_executable(yolo_distance_node ${YOLO_SOURCE_DIR}/src/distance_node.cpp)
target_compile_definitions(yolo_distance_node PRIVATE MULTICLASS_YOLO_MODEL_PACKAGE="yolo_distance_runtime")
ament_target_dependencies(yolo_distance_node rclcpp std_msgs ament_index_cpp)
target_link_libraries(yolo_distance_node multiclass_detector depthai::core
  point_cloud::point_cloud_depth_source Threads::Threads nlohmann_json::nlohmann_json)
install(TARGETS yolo_distance_node DESTINATION lib/${PROJECT_NAME})
# Reuse the model already on Jetson; the downloadable overlay contains no weights.
install(FILES
  ${YOLO_SOURCE_DIR}/models/multiclass_yolox_tiny_512x320_holdout10.onnx
  ${YOLO_SOURCE_DIR}/models/multiclass_yolox_tiny_512x320_holdout10.labels.txt
  ${YOLO_SOURCE_DIR}/models/multiclass_yolox_tiny_512x320_holdout10.metadata.json
  DESTINATION share/${PROJECT_NAME}/models)
ament_package()
